Program -> StmtList
StmtList -> Stmt StmtList | ε
Stmt -> id = Expr ; | if ( Cond ) { StmtList } ElseOpt | while ( Cond ) { StmtList } | print ( Expr ) ;
ElseOpt -> else { StmtList } | ε
Cond -> Expr RelOp Expr
RelOp -> < | > | == | !=
Expr -> Term ExprT
ExprT -> + Term ExprT | - Term ExprT | ε
Term -> Factor TermT
TermT -> * Factor TermT | / Factor TermT | ε
Factor -> ( Expr ) | id | num
